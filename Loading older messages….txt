def _json_from_text(value):
    if not isinstance(value, str):
        raise ValueError("The planning model did not return text.")

    value = value.strip()
    value = re.sub(r"^```(?:json)?\s*|\s*```$", "", value, flags=re.I)

    try:
        return json.loads(value)
    except json.JSONDecodeError:
        start = value.find("{")
        end = value.rfind("}")

        if start >= 0 and end > start:
            return json.loads(value[start:end + 1])

        raise


def _normalise_mode(value, settings=None, prefer_settings=True):
    settings = settings if isinstance(settings, dict) else {}

    def parse(candidate):
        if candidate is None:
            return None

        text = str(candidate).strip().lower().replace("_", "-")
        text = re.sub(r"\s+", " ", text)

        if text in {
            "multiplayer", "multi-player", "multi",
            "online multiplayer", "co-op", "cooperative",
            "versus", "online"
        }:
            return "multiplayer"

        if text in {
            "single-player", "singleplayer", "single",
            "solo", "one player"
        }:
            return "single-player"

        return None

    keys = (
        "players",
        "player_mode",
        "playerMode",
        "player_selection",
        "playerSelection"
    )

    if prefer_settings:
        for key in keys:
            result = parse(settings.get(key))

            if result:
                return result

    result = parse(value)

    if result:
        return result

    if not prefer_settings:
        for key in keys:
            result = parse(settings.get(key))

            if result:
                return result

    return "single-player"

def _project_root(module_dir):
    candidates = [module_dir, *list(module_dir.parents)[:5]]
    markers = (
        "app.py",
        "main.py",
        "run.py",
        "pyproject.toml",
        "requirements.txt"
    )

    for candidate in candidates:
        if any((candidate / marker).is_file() for marker in markers):
            return candidate

    for candidate in candidates:
        if any(
            (candidate / name).is_dir()
            for name in ("static", "frontend", "web", "templates")
        ):
            return candidate

    return module_dir.parent


def _integration_html_context(html):
    if not isinstance(html, str):
        return ""

    script_pattern = re.compile(
        r"(<script\b[^>]*>)([\s\S]*?)(</script\s*>)",
        re.IGNORECASE
    )

    bridge_patterns = (
        re.compile(r"(?:window\s*\.\s*)?AIGameMultiplayer\s*=", re.IGNORECASE),
        re.compile(r"(?:window\s*\.\s*)?AIGameRuntime\s*=", re.IGNORECASE),
        re.compile(r"\bAIGB_MULTIPLAYER_HOST\b"),
        re.compile(r"\bAIGB_MULTIPLAYER\b"),
        re.compile(r"\bsendPlayerRoster\b"),
        re.compile(r"\bsendMultiplayerEvent\b"),
    )

    def replace_script(match):
        opening, body, closing = match.groups()
        attributes = opening.lower()

        external_bridge = bool(re.search(
            r"""src\s*=\s*["'][^"']*(?:runtime|multiplayer)[^"']*["']""",
            attributes
        ))

        inline_bridge = any(
            pattern.search(body) for pattern in bridge_patterns
        )

        if external_bridge or inline_bridge:
            return opening + closing

        return match.group(0)

    return script_pattern.sub(replace_script, html)


def _find_actual_player_js():
    module_dir = Path(__file__).resolve().parent
    root = _project_root(module_dir)

    likely_paths = (
        root / "player.js",
        root / "static" / "player.js",
        root / "js" / "player.js",
        root / "frontend" / "player.js",
        module_dir / "player.js",
        module_dir / "static" / "player.js",
    )

    checked = set()

    for path in likely_paths:
        if path in checked or not path.is_file():
            continue

        checked.add(path)

        try:
            if path.stat().st_size > 200000:
                continue

            source = path.read_text(encoding="utf-8")
        except (OSError, UnicodeError):
            continue

        score = (
            source.count("AIGB_MULTIPLAYER_HOST") * 10
            + source.count("AIGB_MULTIPLAYER") * 5
            + source.count("sendPlayerRoster") * 5
            + source.count("sendMultiplayerEvent") * 5
        )

        if score >= 15:
            return path, source

    candidates = []
    visited_dirs = 0

    for current, directories, files in os.walk(root):
        visited_dirs += 1

        if visited_dirs > 2500:
            break

        directories[:] = [
            name for name in directories
            if name not in PROJECT_CONTEXT_EXCLUDED_DIRS
            and not name.startswith(".")
        ]

        if "player.js" not in files:
            continue

        path = Path(current) / "player.js"

        if path in checked:
            continue

        checked.add(path)

        try:
            if path.stat().st_size > 200000:
                continue

            source = path.read_text(encoding="utf-8")
        except (OSError, UnicodeError):
            continue

        score = (
            source.count("AIGB_MULTIPLAYER_HOST") * 10
            + source.count("AIGB_MULTIPLAYER") * 5
            + source.count("sendPlayerRoster") * 5
            + source.count("sendMultiplayerEvent") * 5
        )

        if score >= 15:
            candidates.append((score, path, source))

    if candidates:
        candidates.sort(
            key=lambda item: (item[0], -len(str(item[1]))),
            reverse=True
        )

        _, path, source = candidates[0]
        return path, source

    return None, None


def _find_project_file(filename):
    module_dir = Path(__file__).resolve().parent
    root = _project_root(module_dir)

    likely_paths = (
        module_dir / filename,
        root / filename,
        root / "backend" / filename,
        root / "server" / filename,
        root / "app" / filename,
        root / "src" / filename,
    )

    checked = set()

    for path in likely_paths:
        resolved = path.resolve()

        if resolved in checked or not resolved.is_file():
            continue

        checked.add(resolved)
        return resolved

    candidates = []
    visited_dirs = 0

    for current, directories, filenames in os.walk(root):
        visited_dirs += 1

        if visited_dirs > 2500:
            break

        directories[:] = [
            name for name in directories
            if name not in PROJECT_CONTEXT_EXCLUDED_DIRS
            and not name.startswith(".")
        ]

        if filename not in filenames:
            continue

        path = (Path(current) / filename).resolve()

        if path in checked:
            continue

        checked.add(path)
        candidates.append(path)

    if not candidates:
        return None

    candidates.sort(
        key=lambda path: (
            len(path.relative_to(root).parts),
            str(path).lower()
        )
    )

    return candidates[0]

def _relevant_source_excerpt(source, markers, max_chars):
    source = str(source or "")

    if len(source) <= max_chars:
        return source

    lines = source.splitlines()
    matches = [
        i for i, line in enumerate(lines)
        if any(marker in line for marker in markers)
    ]

    if not matches:
        return source[:max_chars] + "\n[Source truncated]"

    ranges = []

    for index in matches:
        start = max(0, index - 12)
        end = min(len(lines), index + 26)

        if ranges and start <= ranges[-1][1]:
            ranges[-1] = (ranges[-1][0], max(ranges[-1][1], end))
        else:
            ranges.append((start, end))

    result = ""

    for start, end in ranges:
        section = (
            f"\n[Source lines {start + 1}-{end}]\n"
            + "\n".join(lines[start:end])
            + "\n"
        )

        remaining = max_chars - len(result)

        if remaining <= 0:
            break

        if len(section) > remaining:
            result += section[:remaining]
            break

        result += section

    return result

def _js_tokens(source):
    tokens = []
    i = 0
    brace = 0
    paren = 0
    bracket = 0
    length = len(source)

    while i < length:
        ch = source[i]

        if ch.isspace():
            i += 1
            continue

        if source.startswith("//", i):
            end = source.find("\n", i + 2)
            i = length if end < 0 else end + 1
            continue

        if source.startswith("/*", i):
            end = source.find("*/", i + 2)
            i = length if end < 0 else end + 2
            continue

        if ch in "'\"`":
            quote = ch
            start = i
            i += 1

            while i < length:
                if source[i] == "\\":
                    i += 2
                    continue

                if source[i] == quote:
                    i += 1
                    break

                i += 1

            tokens.append(("<string>", start, i, brace, paren, bracket))
            continue

        if ch.isalpha() or ch in "_$":
            start = i
            i += 1

            while i < length and (
                source[i].isalnum() or source[i] in "_$"
            ):
                i += 1

            tokens.append((source[start:i], start, i, brace, paren, bracket))
            continue

        if ch == "}":
            brace = max(0, brace - 1)
        elif ch == ")":
            paren = max(0, paren - 1)
        elif ch == "]":
            bracket = max(0, bracket - 1)

        tokens.append((ch, i, i + 1, brace, paren, bracket))

        if ch == "{":
            brace += 1
        elif ch == "(":
            paren += 1
        elif ch == "[":
            bracket += 1

        i += 1

    return tokens


def _top_level_declarations(source):
    tokens = _js_tokens(source)
    found = {}

    for index, token in enumerate(tokens):
        kind, _, _, brace, paren, bracket = token

        if kind not in {"let", "var", "const"} or brace or paren or bracket:
            continue

        if index + 1 >= len(tokens):
            continue

        first = tokens[index + 1]

        if not re.fullmatch(r"[A-Za-z_$][\w$]*", first[0]):
            continue

        found[first[0]] = kind
        cursor = index + 2

        while cursor < len(tokens):
            current_kind, _, _, cb, cp, cs = tokens[cursor]

            if (cb, cp, cs) == (0, 0, 0):
                if current_kind == ";" or current_kind in {
                    "let", "var", "const"
                }:
                    break

                if current_kind == "," and cursor + 1 < len(tokens):
                    candidate = tokens[cursor + 1][0]

                    if re.fullmatch(r"[A-Za-z_$][\w$]*", candidate):
                        found[candidate] = kind

            cursor += 1

    return found, tokens


def _adapter_source(declarations):
    mutable = {}
    readonly_objects = {}

    for name, kind in declarations.items():
        if kind in {"let", "var"}:
            mutable[name] = kind
        else:
            readonly_objects[name] = kind

    entries = []

    for name in mutable:
        entries.append(
            f"{json.dumps(name)}:"
            f"{{get:function(){{return {name};}},"
            f"set:function(value){{{name}=value;}},readonly:false}}"
        )

    for name in readonly_objects:
        entries.append(
            f"{json.dumps(name)}:"
            f"{{get:function(){{return {name};}},set:null,readonly:true}}"
        )

    bindings = ",".join(entries)

    return """\n;(function(){

def _has_adapter_call(source):
    for match in re.finditer(
        r"<script\b[^>]*>([\s\S]*?)</script\s*>",
        source,
        re.I
    ):
        tokens = _js_tokens(match.group(1))

        for index in range(len(tokens) - 3):
            if (
                tokens[index][0] == "AIGameRuntime"
                and tokens[index + 1][0] == "."
                and tokens[index + 2][0] == "registerStateAdapter"
                and tokens[index + 3][0] == "("
            ):
                return True

    return False


def _has_top_level_start_call(tokens):
    start_calls = {
        "startGame", "startMatch", "beginGame", "launchGame",
        "initGame", "initializeGame", "runGame", "gameLoop",
        "requestAnimationFrame", "setInterval"
    }

    for index, token in enumerate(tokens[:-1]):
        name, _, _, brace, paren, bracket = token

        if (
            (brace, paren, bracket) != (0, 0, 0)
            or name not in start_calls
            or tokens[index + 1][0] != "("
        ):
            continue

        previous = tokens[index - 1][0] if index else ""

        if previous not in {"function", ".", "?.", "new"}:
            return True

    return False


def _install_automatic_state_adapter(html):
    if (
        'window.__AIGB_AUTO_STATE_ADAPTER__' in html
        or _has_adapter_call(html)
    ):
        return html

    script_pattern = re.compile(
        r"<script\b([^>]*)>([\s\S]*?)</script\s*>",
        re.I
    )

    candidates = []

    for match in script_pattern.finditer(html):
        attrs = match.group(1) or ""

        if re.search(r"\bsrc\s*=", attrs, re.I):
            continue

        if re.search(
            r"\bid\s*=\s*['\"]aigb-(?:runtime|multiplayer)-bridge['\"]",
            attrs,
            re.I
        ):
            continue

        declarations, tokens = _top_level_declarations(match.group(2))

        if declarations:
            candidates.append((match, declarations, tokens, attrs))

    if not candidates:
        return html

    modules = [
        item for item in candidates
        if re.search(r"\btype\s*=\s*['\"]?module\b", item[3], re.I)
    ]

    pool = modules or candidates

    target, declarations, tokens, _ = max(
        pool,
        key=lambda item: (
            int(_has_top_level_start_call(item[2])),
            len(item[1]),
            len(item[0].group(2))
        )
    )

    source = target.group(2)
    insertion = len(source)

    for index, token in enumerate(tokens[:-1]):
        name, position, _, brace, paren, bracket = token

        if (brace, paren, bracket) != (0, 0, 0):
            continue

        if name not in {
            "startGame", "startMatch", "beginGame", "launchGame",
            "initGame", "initializeGame", "runGame", "gameLoop",
            "requestAnimationFrame", "setInterval"
        }:
            continue

        if tokens[index + 1][0] != "(":
            continue

        previous = tokens[index - 1][0] if index else ""

        if previous not in {"function", ".", "?.", "new"}:
            insertion = position
            break

    replacement_body = (
        source[:insertion]
        + _adapter_source(declarations)
        + source[insertion:]
    )

    replacement = target.group(0).replace(source, replacement_body, 1)

    return html[:target.start()] + replacement + html[target.end():]


def _has_inline_game_script(html):
    pattern = re.compile(
        r"<script\b([^>]*)>([\s\S]*?)</script\s*>",
        re.I
    )

    for match in pattern.finditer(html):
        attrs = match.group(1) or ""
        body = match.group(2) or ""

        if re.search(r"\bsrc\s*=", attrs, re.I):
            continue

        if re.search(
            r"\bid\s*=\s*['\"]aigb-(?:runtime|multiplayer)-bridge['\"]",
            attrs,
            re.I
        ):
            continue

        type_match = re.search(
            r"\btype\s*=\s*['\"]?([^'\"\s>]+)",
            attrs,
            re.I
        )

        script_type = type_match.group(1).lower() if type_match else ""

        if script_type not in {
            "",
            "text/javascript",
            "application/javascript",
            "module"
        }:
            continue

        if body.strip():
            return True

    return False


def _attribute_value(attributes, name):
    match = re.search(
        rf"\b{re.escape(name)}\s*=\s*(?:\"([^\"]*)\"|'([^']*)'|([^\s>]+))",
        attributes,
        re.I
    )

    if not match:
        return None

    return next((value for value in match.groups() if value is not None), "")


def _contains_remote_url(value):
    value = str(value or "")
    value = value.replace(r"\/", "/").replace(r"\:", ":")
    value = re.sub(r"&amp;", "&", value, flags=re.I)
    return bool(re.search(r"(?:https?:)?//|\bwww\.", value, re.I))


def _strip_remote_css(css):
    css = re.sub(
        r"@import\s+(?:url\s*\(\s*)?(?:\"[^\"]*\"|'[^']*'|[^;\s)]+)\s*\)?\s*;?",
        lambda match: "" if _contains_remote_url(match.group(0)) else match.group(0),
        css,
        flags=re.I
    )

    def replace_url(match):
        return "none" if _contains_remote_url(match.group(2)) else match.group(0)

    css = re.sub(
        r"url\(\s*(['\"]?)(.*?)\1\s*\)",
        replace_url,
        css,
        flags=re.I | re.S
    )
    return css


def _strip_remote_game_resources(html):
    if not isinstance(html, str) or not html:
        return html

    script_pattern = re.compile(
        r"<script\b([^>]*)>[\s\S]*?</script\s*>",
        re.I
    )

    def clean_script(match):
        attributes = match.group(1) or ""
        src = _attribute_value(attributes, "src")
        return "" if src is not None and _contains_remote_url(src) else match.group(0)

    html = script_pattern.sub(clean_script, html)

    link_pattern = re.compile(r"<link\b[^>]*>", re.I)

    def clean_link(match):
        href = _attribute_value(match.group(0), "href")
        return "" if href is not None and _contains_remote_url(href) else match.group(0)

    html = link_pattern.sub(clean_link, html)

    resource_tag_pattern = re.compile(
        r"<(img|audio|video|source|iframe|embed|track)\b[^>]*>",
        re.I
    )

    def clean_resource_tag(match):
        attributes = match.group(0)
        for attribute in ("src", "href", "poster", "srcset"):
            value = _attribute_value(attributes, attribute)
            if value is not None and _contains_remote_url(value):
                return ""
        return match.group(0)

    html = resource_tag_pattern.sub(clean_resource_tag, html)

    object_pattern = re.compile(r"<object\b[^>]*>", re.I)

    def clean_object(match):
        value = _attribute_value(match.group(0), "data")
        return "" if value is not None and _contains_remote_url(value) else match.group(0)

    html = object_pattern.sub(clean_object, html)

    style_pattern = re.compile(
        r"(<style\b[^>]*>)([\s\S]*?)(</style\s*>)",
        re.I
    )
    html = style_pattern.sub(
        lambda match: match.group(1) + _strip_remote_css(match.group(2)) + match.group(3),
        html
    )

    inline_style_pattern = re.compile(
        r"(\bstyle\s*=\s*)(\"|')([\s\S]*?)\2",
        re.I
    )
    html = inline_style_pattern.sub(
        lambda match: match.group(1) + match.group(2)
        + _strip_remote_css(match.group(3)) + match.group(2),
        html
    )

    return html


def _has_remote_game_resources(html):
    patterns = (
        r"""<script\b[^>]*\bsrc\s*=\s*['\"][^'\"]*(?:https?:)?//""",
        r"""<link\b[^>]*\bhref\s*=\s*['\"][^'\"]*(?:https?:)?//""",
        r"""<(?:img|audio|video|source|iframe|embed)\b[^>]*\b(?:src|href)\s*=\s*['\"][^'\"]*(?:https?:)?//""",
        r"url\(\s*['\"]?\s*(?:https?:)?//",
        r"@import\s+(?:url\()?\s*['\"]?\s*(?:https?:)?//"
    )

    return any(re.search(pattern, html, re.I) for pattern in patterns)


def _extract_complete_html_document(raw):
    if not isinstance(raw, str) or not raw.strip():
        return ""

    text = raw.lstrip("\ufeff").strip()
    doctype_match = re.search(r"<!doctype\s+html\b[^>]*>", text, re.I)
    html_match = re.search(r"<html\b[^>]*>", text, re.I)

    if doctype_match and (html_match is None or doctype_match.start() < html_match.start()):
        start = doctype_match.start()
    elif html_match:
        start = html_match.start()
    else:
        return ""

    opening_html = re.search(r"<html\b[^>]*>", text[start:], re.I)
    if not opening_html:
        return ""

    scan_start = start + opening_html.end()
    tag_pattern = re.compile(r"<script\b[^>]*>|</script\s*>|</html\s*>", re.I)
    inside_script = False

    for match in tag_pattern.finditer(text, scan_start):
        tag = match.group(0)
        lowered = tag.lower()

        if lowered.startswith("<script"):
            if not lowered.rstrip().endswith("/>"):
                inside_script = True
            continue

        if lowered.startswith("</script"):
            inside_script = False
            continue

        if lowered.startswith("</html") and not inside_script:
            return text[start:match.end()].strip()

    return ""


def _html_document_issues(raw, require_script=False):
    html = _extract_complete_html_document(raw)
    if not html:
        return ["No complete HTML document was found. Return one document from <!DOCTYPE html> through </html>."]

    html = _strip_remote_game_resources(html)
    issues = []

    if not re.search(r"<html\b", html, re.I):
        issues.append("The document is missing its html element.")
    if not re.search(r"<body\b", html, re.I):
        issues.append("The document is missing its body element.")
    if not re.search(r"</html\s*>$", html, re.I):
        issues.append("The document does not end with </html>.")
    if _has_remote_game_resources(html):
        issues.append("Remote scripts or asset URLs remain after platform-safe cleanup.")
    if require_script and not _has_inline_game_script(html):
        issues.append("The multiplayer document has no inline game script.")

    return issues


def _extract_valid_game_html(raw, require_script=False):
    html = _extract_complete_html_document(raw)
    if not html:
        return ""

    html = _strip_remote_game_resources(html)
    if _html_document_issues(html, require_script=require_script):
        return ""

    return html

def _resource_catalogue_for_request(prompt, plan):

    )

    plan_text = json.dumps(plan, ensure_ascii=False)

    if not resource_terms.search(str(prompt) + "\n" + plan_text):

        )


def _prepare_html(raw, mode):
    html = _extract_complete_html_document(raw)

    if not isinstance(html, str) or not html.strip():
        raise ValueError("Ollama did not return one complete HTML document from <!DOCTYPE html> through </html>.")

    html = _strip_remote_game_resources(html.strip())

    if not re.search(r"<html\b", html, re.I):
        raise ValueError("The generated game is missing its HTML document.")

    if not re.search(r"<body\b", html, re.I):
        raise ValueError("The generated game is missing its HTML body.")

    if not re.search(r"</html\s*>", html, re.I):
        raise ValueError("The generated game document is incomplete.")

    if _has_remote_game_resources(html):
        raise ValueError(
            "The generated game still contains an external resource URL after cleanup."
        )

    if mode == "multiplayer":
        html = _remove_placeholder_room_bootstrap(html)
        html = _remove_zero_arg_global_event_functions(html)
        html = _remove_bad_multiplayer_subscriptions(html)
        if not _has_inline_game_script(html):
            raise ValueError(
                "The multiplayer game has no usable inline game script."
            )

        html = attach_multiplayer(html)

    html = attach_runtime(html)
    html = _install_automatic_state_adapter(html)

    return html


def _game_script_bodies(html):
    pattern = re.compile(

    )

    for match in pattern.finditer(html):
        attrs = match.group(1) or ""
        body = match.group(2) or ""

        if re.search(r"\bsrc\s*=", attrs, re.I):
            continue

        if re.search(
            r"\bid\s*=\s*['\"]aigb-(?:runtime|multiplayer)-bridge['\"]",
            attrs,
            re.I
        ):
            continue

        type_match = re.search(
            r"\btype\s*=\s*['\"]?([^'\"\s>]+)",
            attrs,
            re.I
        )

        script_type = type_match.group(1).lower() if type_match else ""

        if script_type not in {
            "",
            "text/javascript",
            "application/javascript",
            "module"
        }:
            continue

        if body.strip():
            yield body


def _has_connected_state_guard(scripts, targets):
    state_sends = list(re.finditer(
        targets + r"\s*\.\s*send\s*\(\s*(['\"])state\1\s*,",
        scripts,
        re.I
    ))

    if not state_sends:
        return False

    negative_guard = re.compile(
        r"if\s*\(\s*(?:!\s*" + targets
        + r"\s*\.\s*connected|" + targets
        + r"\s*\.\s*connected\s*!==?\s*true)"
        + r"(?:\s*\|\|\s*[^)]*)?\s*\)"
        + r"\s*(?:\{\s*)?return\b",
        re.I
    )
    positive_guard = re.compile(
        r"if\s*\(\s*" + targets
        + r"\s*\.\s*connected(?:\s*===?\s*true)?"
        + r"(?:\s*&&\s*[^)]*)?\s*\)\s*\{[^{}]{0,1200}$",
        re.I
    )

    for send_match in state_sends:
        prefix = scripts[max(0, send_match.start() - 2200):send_match.start()]
        if negative_guard.search(prefix) or positive_guard.search(prefix):
            return True

    return False


def _multiplayer_api_status(html):
    scripts = "\n".join(_game_script_bodies(html))
    aliases = [r"(?:window\s*\.)?AIGameMultiplayer"]

    alias_pattern = re.compile(
        r"\b(?:const|let|var)\s+([A-Za-z_$][\w$]*)\s*=\s*(?:window\s*\.)?AIGameMultiplayer\b"
    )

    for match in alias_pattern.finditer(scripts):
        aliases.append(re.escape(match.group(1)))

    targets = r"(?:" + "|".join(aliases) + r")"
    on_calls = re.findall(
        targets + r"\s*\.\s*on\s*\(\s*(['\"])(.*?)\1",
        scripts,
        re.I
    )
    send_calls = re.findall(
        targets + r"\s*\.\s*send\s*\(\s*(['\"])(.*?)\1",
        scripts,
        re.I
    )

    subscribed = {name for _, name in on_calls}
    sent = {name for _, name in send_calls}
    subscribed_lower = {name.lower() for name in subscribed}
    sent_lower = {name.lower() for name in sent}

    allowed_subscriptions = {"connection", "players", "player_joined", "state"}
    allowed_sends = {"state"}
    unsupported_subscriptions = sorted(
        name for name in subscribed if name.lower() not in allowed_subscriptions
    )
    unsupported_sends = sorted(
        name for name in sent if name.lower() not in allowed_sends
    )
    sent_host_events = sorted(
        name for name in sent if name.lower() in {"connection", "players"}
    )

    player_id_reference = bool(re.search(
        targets + r"\s*\.\s*playerId\b",
        scripts,
        re.I
    ))

    state_send_calls = list(re.finditer(
        targets + r"\s*\.\s*send\s*\(\s*(['\"])state\1\s*,",
        scripts,
        re.I
    ))

    payload_has_player_id = False
    if state_send_calls:
        player_id_property = re.compile(
            r"\bplayerId\s*:\s*(?:(?:window\s*\.)?[A-Za-z_$][\w$]*\s*\.\s*)*playerId\b",
            re.I
        )
        direct_player_id_property = re.compile(r"\bplayerId\s*:", re.I)

        for send_match in state_send_calls:
            payload_start = send_match.end()
            payload_end = _matching_delimiter(
                scripts,
                scripts.find("(", send_match.start()),
                "(",
                ")"
            )
            if payload_end < 0:
                continue

            arguments = scripts[payload_start:payload_end].strip()
            payload_sources = [arguments]

            variable_match = re.match(r"([A-Za-z_$][\w$]*)\b", arguments)
            if variable_match:
                variable_name = re.escape(variable_match.group(1))
                declaration_patterns = (
                    rf"\b(?:const|let|var)\s+{variable_name}\s*=\s*\{{([\s\S]*?)\}}",
                    rf"\b{variable_name}\s*=\s*\{{([\s\S]*?)\}}"
                )
                for declaration_pattern in declaration_patterns:
                    for declaration in re.finditer(declaration_pattern, scripts):
                        payload_sources.append(declaration.group(1))

            for payload_source in payload_sources:
                if direct_player_id_property.search(payload_source) and (
                    player_id_property.search(payload_source)
                    or re.search(r"\bplayerId\s*:\s*playerId\b", payload_source)
                ):
                    payload_has_player_id = True
                    break

            if payload_has_player_id:
                break

    bridge_connected_reference = bool(re.search(
        targets + r"\s*\.\s*connected\b",
        scripts,
        re.I
    ))
    bridge_connected_guard = _has_connected_state_guard(scripts, targets)

    return {
        "roster_subscription": "players" in subscribed_lower,
        "connection_subscription": "connection" in subscribed_lower,
        "state_subscription": "state" in subscribed_lower,
        "gameplay_subscription": "state" in subscribed_lower,
        "gameplay_send": "state" in sent_lower,
        "matching_gameplay_event": "state" in subscribed_lower and "state" in sent_lower,
        "unsupported_event_subscriptions": unsupported_subscriptions,
        "unsupported_event_sends": unsupported_sends,
        "sent_host_events": sent_host_events,
        "player_id_reference": player_id_reference,
        "bridge_connected_reference": bridge_connected_reference,
        "bridge_connected_guard": bridge_connected_guard,
        "outgoing_player_id": payload_has_player_id,
        "subscribed_events": sorted(subscribed),
        "sent_events": sorted(sent)
    }


def _multiplayer_gameplay_issues(html, original_request="", plan=None):
    scripts = "\n".join(_game_script_bodies(html))
    issues = []
    executable_tokens = [token[0] for token in _js_tokens(scripts)]

    forbidden_network_apis = (
        "WebSocket", "XMLHttpRequest", "EventSource", "sendBeacon", "fetch"
    )
    for identifier in forbidden_network_apis:
        if identifier in executable_tokens:
            issues.append(
                "game uses forbidden direct network API: " + identifier
                + ". Use only the actual installed AAI multiplayer bridge."
            )

    if re.search(r"(?<![\w$.])io\s*\(", scripts):
        issues.append("game initializes a separate Socket.IO connection instead of using the platform bridge")

    if re.search(r"your[-_ ]room[-_ ]code|\b(?:connectToRoom|createRoom|joinRoom|leaveRoom|reconnectToRoom)\s*\(", scripts, re.I):
        issues.append("game invents its own room-management function or placeholder room code")

    placeholder_patterns = (
        r"update\s+player\s+positions",
        r"initialize\s+game\s+logic",
        r"render\s+game\s+entities\s+through",
        r"implement\s+.*(?:movement|render|synchron|gameplay)",
        r"handle\s+joining,?\s+leaving",
        r"check\s+for\s+collisions.*boundar",
        r"broadcast\s+local\s+player.*position",
        r"receive\s+and\s+apply\s+other\s+players"
    )
    comment_text = "\n".join(re.findall(r"//[^\n]*|/\*[\s\S]*?\*/", scripts))
    for marker in placeholder_patterns:
        if re.search(marker, comment_text, re.I):
            issues.append("game contains placeholder comments describing missing gameplay implementation")
            break

    keydown_listener = bool(re.search(
        r"(?:addEventListener|on)\s*\(\s*(['\"])keydown\1",
        scripts,
        re.I
    )) or bool(re.search(r"\bonkeydown\s*=", scripts, re.I))
    keyup_listener = bool(re.search(
        r"(?:addEventListener|on)\s*\(\s*(['\"])keyup\1",
        scripts,
        re.I
    )) or bool(re.search(r"\bonkeyup\s*=", scripts, re.I))

    requested_text = str(original_request or "") + " " + json.dumps(plan or {}, ensure_ascii=False)
    requests_wasd = bool(re.search(r"\bW\s*[,/]\s*A\s*[,/]\s*S\s*[,/]\s*D\b|\bWASD\b", requested_text, re.I))
    if requests_wasd and (not keydown_listener or not keyup_listener):
        issues.append("the requested WASD controls do not have both real keydown and keyup handlers")

    if re.search(r"\b(?:event|e|ev)\s*\.\s*key\b", scripts, re.I) and not keydown_listener:
        issues.append("keyboard code reads a key event without a registered keydown handler")

    if re.search(r"<canvas\b", html, re.I):
        draw_operation = bool(re.search(
            r"\.\s*(?:fillRect|strokeRect|drawImage|clearRect|fillText|arc|fill|stroke|roundRect)\s*\(",
            scripts,
            re.I
        ))
        animation_loop = bool(re.search(
            r"requestAnimationFrame\s*\(|setInterval\s*\(",
            scripts,
            re.I
        ))
        if not draw_operation:
            issues.append("canvas game has no actual drawing operations")
        if not animation_loop:
            issues.append("canvas game has no persistent update or rendering loop")

        function_pattern = re.compile(
            r"\bfunction\s+(render|draw|drawGame|drawScene|renderGame|startGame|startMatch|updatePlayerPositions|updateGame|gameLoop|movePlayer|initializeGame|initGame)\s*\([^)]*\)\s*\{",
            re.I
        )
        empty_names = []
        for match in function_pattern.finditer(scripts):
            open_index = scripts.find("{", match.start(), match.end())
            close_index = _matching_delimiter(scripts, open_index, "{", "}")
            if close_index < 0:
                continue
            body_tokens = [
                token[0] for token in _js_tokens(scripts[open_index + 1:close_index])
                if token[0] not in {";"}
            ]
            if not body_tokens:
                empty_names.append(match.group(1))
        if empty_names:
            issues.append(
                "game contains empty gameplay functions instead of implementations: "
                + ", ".join(sorted(set(empty_names)))
            )

        renderer_match = re.search(
            r"\bfunction\s+(renderGame|drawGame|render|draw)\s*\(\s*([A-Za-z_$][\w$]*)\s*\)\s*\{",
            scripts,
            re.I
        )
        if renderer_match:
            render_name = renderer_match.group(1)
            parameter = renderer_match.group(2)
            open_index = scripts.find("{", renderer_match.start(), renderer_match.end())
            close_index = _matching_delimiter(scripts, open_index, "{", "}")
            body = scripts[open_index + 1:close_index] if close_index >= 0 else ""
            expects_object = bool(re.search(
                rf"\b{re.escape(parameter)}\s*\.\s*(?:players|entities|objects|state)\b",
                body
            ))
            direct_raf = bool(re.search(
                rf"requestAnimationFrame\s*\(\s*{re.escape(render_name)}\s*\)",
                scripts
            ))
            if expects_object and direct_raf:
                issues.append(
                    "animation loop passes the browser timestamp directly into "
                    + render_name + " even though that function expects a game-state object"
                )

    local_remote_assignment = re.search(
        r"\b(?:playerX|playerY|playerColor|localPlayerX|localPlayerY|localPlayerColor)\s*=\s*(?:data|payload|state|remote)\s*\.\s*(?:x|y|color)",
        scripts,
        re.I
    ) or re.search(
        r"\blocalPlayer\s*\.\s*(?:x|y|color)\s*=\s*(?:data|payload|state|remote)\s*\.",
        scripts,
        re.I
    )
    if local_remote_assignment:
        issues.append("incoming remote state overwrites local-player state")

    roster_match = re.search(
        r"\.on\s*\(\s*(['\"])players\1\s*,\s*(?:function\s*\(\s*([A-Za-z_$][\w$]*)\s*\)|\(\s*([A-Za-z_$][\w$]*)\s*\)\s*=>|([A-Za-z_$][\w$]*)\s*=>)\s*\{([\s\S]{0,4000}?)\}\s*\)",
        scripts,
        re.I
    )
    if roster_match:
        roster_body = roster_match.group(5)
        if not re.search(r"remote|playersBy|roster|\.set\s*\(|\.delete\s*\(|\.map\s*\(|forEach\s*\(|for\s*\(|=\s*[A-Za-z_$]", roster_body, re.I):
            issues.append("players roster event is received but not used to maintain the remote-player list")

    state_send = re.search(r"\.send\s*\(\s*(['\"])state\1", scripts, re.I)
    if state_send:
        throttle_markers = re.search(
            r"setInterval\s*\(|setTimeout\s*\(|Date\.now\s*\(|performance\.now\s*\(|last(?:State)?Send|lastBroadcast|sendThrottle|sendInterval|nextSendAt|sendStateAt|lastStatePublishAt",
            scripts,
            re.I
        )
        if not throttle_markers:
            issues.append("game sends state without a visible rate limit or throttled publisher")

    return issues

def _strip_function_declaration(source, function_name):
    pattern = re.compile(
        rf"\bfunction\s+{re.escape(function_name)}\s*\([^)]*\)\s*\{{",
        re.I
    )
    match = pattern.search(source)

    if not match:
        return source

    open_index = source.find("{", match.start(), match.end())
    depth = 0
    quote = None
    i = open_index

    while i < len(source):
        ch = source[i]

        if quote:
            if ch == "\\":
                i += 2
                continue
            if ch == quote:
                quote = None
            i += 1
            continue

        if source.startswith("//", i):
            newline = source.find("\n", i + 2)
            i = len(source) if newline < 0 else newline + 1
            continue

        if source.startswith("/*", i):
            close = source.find("*/", i + 2)
            i = len(source) if close < 0 else close + 2
            continue

        if ch in "'\"`":
            quote = ch
            i += 1
            continue

        if ch == "{":
            depth += 1
        elif ch == "}":
            depth -= 1
            if depth == 0:
                return source[:match.start()] + source[i + 1:]
        i += 1

    return source


def _matching_delimiter(source, open_index, opening, closing):
    depth = 0
    quote = None
    i = open_index

    while i < len(source):
        ch = source[i]

        if quote:
            if ch == "\\":
                i += 2
                continue
            if ch == quote:
                quote = None
            i += 1
            continue

        if source.startswith("//", i):
            newline = source.find("\n", i + 2)
            i = len(source) if newline < 0 else newline + 1
            continue

        if source.startswith("/*", i):
            close = source.find("*/", i + 2)
            i = len(source) if close < 0 else close + 2
            continue

        if ch in "'\"`":
            quote = ch
            i += 1
            continue

        if ch == opening:
            depth += 1
        elif ch == closing:
            depth -= 1
            if depth == 0:
                return i
        i += 1

    return -1


def _remove_zero_arg_global_event_functions(html):
    script_pattern = re.compile(
        r"(<script\b([^>]*)>)([\s\S]*?)(</script\s*>)",
        re.I
    )

    def clean_script(match):
        attrs = match.group(2) or ""
        body = match.group(3) or ""

        if re.search(r"\bsrc\s*=", attrs, re.I):
            return match.group(0)

        if re.search(
            r"\bid\s*=\s*['\"]aigb-(?:runtime|multiplayer)-bridge['\"]",
            attrs,
            re.I
        ):
            return match.group(0)

        names = re.findall(
            r"\bfunction\s+([A-Za-z_$][\w$]*)\s*\(\s*\)\s*\{",
            body
        )

        for name in names:
            declaration = re.search(
                rf"\bfunction\s+{re.escape(name)}\s*\(\s*\)\s*\{{",
                body
            )
            if not declaration:
                continue

            open_brace = body.find("{", declaration.start(), declaration.end())
            close_brace = _matching_delimiter(body, open_brace, "{", "}")
            if close_brace < 0:
                continue

            function_body = body[open_brace + 1:close_brace]
            if not re.search(r"\bevent\s*\.\s*(?:key|code)\b", function_body):
                continue

            invoked = re.search(
                rf"(?:^|[;\n])\s*{re.escape(name)}\s*\(\s*\)\s*;",
                body,
                re.M
            ) or re.search(
                rf"requestAnimationFrame\s*\(\s*{re.escape(name)}\s*\)",
                body
            )

            if not invoked:
                continue

            body = _strip_function_declaration(body, name)
            body = re.sub(
                rf"(?<![\w$.]){re.escape(name)}\s*\(\s*\)\s*;?",
                "",
                body
            )
            body = re.sub(
                rf"requestAnimationFrame\s*\(\s*{re.escape(name)}\s*\)\s*;?",
                "",
                body
            )

        return match.group(1) + body + match.group(4)

    return script_pattern.sub(clean_script, html)

def _remove_bad_multiplayer_subscriptions(html):
    script_pattern = re.compile(
        r"(<script\b([^>]*)>)([\s\S]*?)(</script\s*>)",
        re.I
    )
    subscription_pattern = re.compile(
        r"(?:window\s*\.\s*)?AIGameMultiplayer\s*\.\s*on\s*\(\s*(['\"])(players|state)\1\s*,",
        re.I
    )

    def clean_script(match):
        attrs = match.group(2) or ""
        body = match.group(3) or ""

        if re.search(r"\bsrc\s*=", attrs, re.I):
            return match.group(0)

        if re.search(
            r"\bid\s*=\s*['\"]aigb-(?:runtime|multiplayer)-bridge['\"]",
            attrs,
            re.I
        ):
            return match.group(0)

        changed = True
        while changed:
            changed = False

            for subscription in subscription_pattern.finditer(body):
                event_name = subscription.group(2).lower()
                call_open = body.find("(", subscription.start(), subscription.end())
                call_close = _matching_delimiter(body, call_open, "(", ")")

                if call_close < 0:
                    continue

                callback_source = body[subscription.end():call_close]
                signature = re.match(
                    r"\s*(?:function(?:\s+[A-Za-z_$][\w$]*)?\s*\(\s*([A-Za-z_$][\w$]*)\s*\)|\(?\s*([A-Za-z_$][\w$]*)\s*\)?\s*=>)\s*\{",
                    callback_source
                )

                if not signature:
                    continue

                parameter = signature.group(1) or signature.group(2)
                open_brace = body.find("{", subscription.end(), call_close)
                close_brace = _matching_delimiter(body, open_brace, "{", "}")

                if open_brace < 0 or close_brace < 0 or close_brace > call_close:
                    continue

                callback_body = body[open_brace + 1:close_brace]
                should_remove = False

                if event_name == "state":
                    local_assignment = re.search(
                        rf"\b(?:playerX|playerY|playerColor|localPlayerX|localPlayerY|localPlayerColor)\s*=\s*{re.escape(parameter)}\s*\.\s*(?:x|y|color)",
                        callback_body,
                        re.I
                    ) or re.search(
                        rf"\blocalPlayer\s*\.\s*(?:x|y|color)\s*=\s*{re.escape(parameter)}\s*\.",
                        callback_body,
                        re.I
                    )
                    should_remove = bool(local_assignment)

                if event_name == "players":
                    roster_use = re.search(
                        r"remote|playersBy|roster|\.set\s*\(|\.delete\s*\(|\.map\s*\(|forEach\s*\(|for\s*\(|=\s*[A-Za-z_$]",
                        callback_body,
                        re.I
                    )
                    should_remove = not bool(roster_use)

                if should_remove:
                    remove_end = call_close + 1
                    while remove_end < len(body) and body[remove_end].isspace():
                        remove_end += 1
                    if remove_end < len(body) and body[remove_end] == ";":
                        remove_end += 1
                    body = body[:subscription.start()] + body[remove_end:]
                    changed = True
                    break

        return match.group(1) + body + match.group(4)

    return script_pattern.sub(clean_script, html)

def _remove_placeholder_room_bootstrap(html):
    script_pattern = re.compile(

    )

    def clean_script(match):

    return script_pattern.sub(clean_script, html)

def _multiplayer_patch_issues(patch):
    tokens = _js_tokens(patch)
    names = [token[0] for token in tokens]
    issues = []

    forbidden_identifiers = {
        "postMessage": "direct postMessage usage",
        "fetch": "direct fetch usage",
        "XMLHttpRequest": "direct XMLHttpRequest usage",
        "WebSocket": "direct WebSocket usage",
        "EventSource": "direct EventSource usage",
        "sendBeacon": "direct sendBeacon usage"
    }

    for identifier, label in forbidden_identifiers.items():
        if identifier in names:
            issues.append(label)

    if re.search(r"\bwindow\s*\.\s*(?:parent|top)\b", patch):
        issues.append("direct parent or top frame access")

    if re.search(r"(?<![\w$.])parent\s*\.", patch):
        issues.append("direct parent-frame access")

    if re.search(r"(?<![\w$.])top\s*\.", patch):
        issues.append("direct top-frame access")

    if re.search(r"\baddEventListener\s*\(\s*(['\"])message\1", patch, re.I):
        issues.append("direct host message listener")

    if "AIGB_MULTIPLAYER" in patch:
        issues.append("private multiplayer channel reference")

    if re.search(r"\b(?:connectToRoom|createRoom|joinRoom|leaveRoom)\s*\(", patch):
        issues.append("invented room creation or joining function")

    if re.search(r"your[-_ ]room[-_ ]code|placeholder[-_ ]room", patch, re.I):
        issues.append("placeholder room code")

    return list(dict.fromkeys(issues))


def _extract_multiplayer_patch(raw):
    if not isinstance(raw, str) or not raw.strip():
        raise ValueError("Ollama returned an empty multiplayer integration patch.")

    patch = raw.strip()

    fenced = re.fullmatch(
        r"\s*```(?:javascript|js)?\s*([\s\S]*?)\s*```\s*",
        patch,
        re.I
    )

    if fenced:
        patch = fenced.group(1).strip()

    script_match = re.fullmatch(
        r"\s*<script\b[^>]*>([\s\S]*?)</script\s*>\s*",
        patch,
        re.I
    )

    if script_match:
        patch = script_match.group(1).strip()

    if not patch:
        raise ValueError("Ollama returned an empty JavaScript patch.")

    if re.search(r"</?html\b|</?body\b|</?head\b", patch, re.I):
        raise ValueError(
            "Ollama returned HTML instead of a JavaScript integration patch."
        )

    closing_tags = list(re.finditer(r"</script\s*>", patch, re.I))

    if closing_tags:
        tokens = _js_tokens(patch)
        string_ranges = [
            (token[1], token[2])
            for token in tokens
            if token[0] == "<string>"
        ]
        replacements = []
        removable_terminal = []

        for match in closing_tags:
            inside_string = any(
                start <= match.start() < end
                for start, end in string_ranges
            )

            if inside_string:
                replacements.append(match)
            elif (
                match is closing_tags[-1]
                and not patch[match.end():].strip()
                and not re.search(r"<script\b", patch[:match.start()], re.I)
            ):
                removable_terminal.append(match)
            else:
                raise ValueError(
                    "The multiplayer patch contains HTML markup instead of "
                    "JavaScript. The patch was rejected before insertion."
                )

        for match in reversed(replacements):
            patch = (
                patch[:match.start()]
                + re.sub(r"</script", r"<\/script", match.group(0), flags=re.I)
                + patch[match.end():]
            )

        for match in reversed(removable_terminal):
            patch = patch[:match.start()].rstrip()

    if not patch.strip():
        raise ValueError("Ollama returned an empty JavaScript integration patch.")

    return patch.strip()

def _inject_multiplayer_patch(html, patch):
    script_pattern = re.compile(
        r"(<script\b([^>]*)>)([\s\S]*?)(</script\s*>)",
        re.I
    )

    candidates = []

    for match in script_pattern.finditer(html):
        attrs = match.group(2) or ""
        body = match.group(3) or ""

        if re.search(r"\bsrc\s*=", attrs, re.I):
            continue

        if re.search(
            r"\bid\s*=\s*['\"]aigb-(?:runtime|multiplayer)-bridge['\"]",
            attrs,
            re.I
        ):
            continue

        type_match = re.search(
            r"\btype\s*=\s*['\"]?([^'\"\s>]+)",
            attrs,
            re.I
        )

        script_type = type_match.group(1).lower() if type_match else ""

        if script_type not in {
            "",
            "text/javascript",
            "application/javascript",
            "module"
        }:
            continue

        if not body.strip():
            continue

        declarations, tokens = _top_level_declarations(body)

        score = (
            int(_has_top_level_start_call(tokens)),
            len(declarations),
            len(body)
        )

        candidates.append((score, match))

    if not candidates:
        raise ValueError(
            "Cannot integrate multiplayer: the generated HTML contains "
            "no usable inline JavaScript game script."
        )

    target = max(candidates, key=lambda item: item[0])[1]
    body = target.group(3)
    integration = "\n;\n(function(){\n" + patch + "\n})();\n"
    updated_body = body + integration

    return (
        html[:target.start(3)]
        + updated_body
        + html[target.end(3):]
    )


def _remove_platform_bridge_script_tags(html):
    pattern = re.compile(
        r"<script\b([^>]*)>[\s\S]*?</script\s*>",
        re.I
    )

    def remove_bridge(match):
        attributes = match.group(1) or ""
        bridge_id = re.search(
            r"\bid\s*=\s*(['\"])aigb-(?:runtime|multiplayer)-bridge\1",
            attributes,
            re.I
        )
        return "" if bridge_id else match.group(0)

    return pattern.sub(remove_bridge, html)

def _call_html_model(
    messages,
    user_payload,
    mode,
    stage,
    progress=None,
    temperature=0.2,
    stream_progress=None,
    extra_system_messages=None,
    model=None
):
    raw = chat(
        messages,
        temperature=temperature,
        timeout=3600,
        progress=stream_progress,
        model=model
    )

    html = _extract_valid_game_html(
        raw,
        require_script=(mode == "multiplayer")
    )

    if html:
        return html

    first_response_issues = _html_document_issues(
        raw,
        require_script=(mode == "multiplayer")
    )

    if progress:
        progress(
            78,
            "Retrying game generation",
            "The first response did not produce acceptable standalone HTML. Sending the rejected response and exact problems back to Ollama."
        )

    retry_messages = [
        {
            "role": "system",
            "content": SYSTEM_HTML_RETRY + "\n\n" + SYSTEM_NO_COMMENTS
        }
    ]

    if mode == "multiplayer":
        retry_messages.append({
            "role": "system",
            "content": MULTIPLAYER_API_CONTRACT
        })

    for message in extra_system_messages or []:
        retry_messages.append(message)

    retry_payload = {
        "task": (
            "Repair the rejected response and return the requested game as "
            "one complete standalone HTML document."
        ),
        "full_original_task_context": user_payload,
        "rejected_response_excerpt": str(raw or "")[-18000:],
        "detected_problems": first_response_issues or [
            "The response could not be extracted as one complete HTML document."
        ],
        "retry_requirements": {
            "extract_only_the_document_from_doctype_or_html_through_closing_html": True,
            "follow_the_original_request_exactly": True,
            "preserve_the_requested_game_mechanics": True,
            "do_not_replace_the_requested_game_with_a_sample": True,
            "return_one_complete_html_document": True,
            "put_css_and_javascript_inside_the_html_document": True,
            "do_not_return_code_fences_or_explanations": True,
            "do_not_use_remote_resource_urls": True,
            "use_only_the_platform_installed_runtime_and_multiplayer_bridge": True
        }
    }

    retry_messages.append({
        "role": "user",
        "content": json.dumps(retry_payload, ensure_ascii=False)
    })

    retry_raw = chat(
        retry_messages,
        temperature=0.1,
        timeout=3600,
        model=model
    )

    html = _extract_valid_game_html(
        retry_raw,
        require_script=(mode == "multiplayer")
    )

    if not html:
        retry_issues = _html_document_issues(
            retry_raw,
            require_script=(mode == "multiplayer")
        )
        raise ValueError(
            stage
            + " failed after two attempts: "
            + "; ".join(retry_issues or ["The response was not a complete usable HTML document."])
            + ". The invalid response was stopped before runtime preparation, "
            "multiplayer integration, or saving."
        )

    return html


def plan_game(prompt, settings, progress=None, model=None):
    settings = settings if isinstance(settings, dict) else {}
    model = model or _model_for_settings(settings)

    if progress:
        progress(
            8,
            "Planning the game",
            "Ollama is turning your prompt into a game design."
        )

    plan_payload = {
        "request": prompt,
        "settings": settings
    }

    messages = [
        {
            "role": "system",
            "content": SYSTEM_PLAN
        },
        {
            "role": "user",
            "content": json.dumps(plan_payload, ensure_ascii=False)
        }
    ]

    content = chat(
        messages,
        json_mode=True,
        temperature=0.15,
        timeout=3600,
        model=model
    )

    try:
        plan = _json_from_text(content)
    except (ValueError, json.JSONDecodeError):
        retry_content = chat(
            [
                {
                    "role": "system",
                    "content": (
                        SYSTEM_PLAN
                        + "\n\nYour previous response was not valid JSON. "
                        "Return one valid JSON object only."
                    )
                },
                {
                    "role": "user",
                    "content": json.dumps(plan_payload, ensure_ascii=False)
                }
            ],
            json_mode=True,
            temperature=0.0,
            timeout=3600,
            model=model
        )

        plan = _json_from_text(retry_content)

    if not isinstance(plan, dict):
        raise ValueError("The planning model did not return a JSON game plan.")

    plan["player_mode"] = _normalise_mode(
        plan.get("player_mode"),
        settings
    )

    if plan["player_mode"] == "single-player":
        plan["multiplayer_design"] = (
            "Not applicable. This is a single-player game."
        )

    if progress:
        progress(
            28,
            "Game plan ready",
            "The design plan is ready. Preparing the game source."
        )

    return plan


def _validate_or_raise(html, label):
    result = validate_game(html, require_runtime=True)

    if not isinstance(result, dict):
        raise ValueError(label + ": the validator returned an invalid result.")

    if not result.get("valid"):
        errors = result.get("errors") or ["Unknown validation error"]

        raise ValueError(
            label + ": " + "; ".join(str(error) for error in errors)
        )

    return result


def _validate_with_ai_repair(
    html,
    label,
    original_request,
    settings,
    plan,
    mode,
    progress=None,
    platform_context=None,
    stage="generation",
    model=None
):
    model = model or _model_for_settings(settings)
    try:
        return html, _validate_or_raise(html, label)
    except ValueError as initial_error:
        if progress:
            progress(
                92,
                "Repairing validation errors",
                "The platform validator found specific problems. Sending those errors and the current HTML to Ollama for one repair attempt."
            )

        repair_messages = [
            {
                "role": "system",
                "content": SYSTEM_QA + "\n\n" + SYSTEM_NO_COMMENTS
            }
        ]

        if mode == "multiplayer":
            if platform_context is None:
                platform_context = _load_platform_project_context(
                    require_multiplayer=True
                )
            repair_messages.append({
                "role": "system",
                "content": MULTIPLAYER_API_CONTRACT
            })
            repair_messages.append({
                "role": "system",
                "content": _multiplayer_integration_platform_message(platform_context)
            })

        repair_payload = {
            "task": "Repair the exact current game so it passes the AAI Game Builder validator. Return one complete HTML document.",
            "stage": stage,
            "original_user_request": original_request,
            "settings": settings,
            "game_design_plan": plan or {},
            "validator_errors": str(initial_error),
            "current_complete_html": html,
            "repair_rules": [
                "Fix the listed validator errors, not unrelated features.",
                "Preserve the requested game, its visual identity, mechanics, controls, and working behavior.",
                "Return one complete HTML document from <!DOCTYPE html> through </html>.",
                "Keep CSS and JavaScript inside the HTML document.",
                "Do not return a patch, Markdown, code fences, explanations, or comments.",
                "Do not use remote scripts, CDNs, remote assets, or external dependencies."
            ]
        }

        try:
            raw = chat(
                repair_messages + [{
                    "role": "user",
                    "content": json.dumps(repair_payload, ensure_ascii=False)
                }],
                temperature=0.0,
                timeout=3600,
                model=model
            )

            repaired_html = _extract_valid_game_html(
                raw,
                require_script=(mode == "multiplayer")
            )
            if not repaired_html:
                issues = _html_document_issues(
                    raw,
                    require_script=(mode == "multiplayer")
                )
                raise ValueError(
                    "Ollama's validation repair did not return usable HTML: "
                    + "; ".join(issues)
                )

            if mode == "multiplayer":
                repaired_html = _remove_platform_bridge_script_tags(repaired_html)

            repaired_html = _prepare_html(repaired_html, mode)

            if mode == "multiplayer":
                repaired_html = _ensure_multiplayer_gameplay(
                    repaired_html,
                    original_request,
                    settings,
                    plan,
                    progress,
                    stage + "-validation-repair",
                    platform_context,
                    model=model
                )

            validation = _validate_or_raise(
                repaired_html,
                label + " after AI repair"
            )
            return repaired_html, validation

        except Exception as repair_error:
            raise ValueError(
                str(initial_error)
                + ". Ollama was given one repair attempt, but the repaired game "
                + "still did not pass validation: "
                + str(repair_error)
            ) from repair_error

    settings = _load_game_settings(game)
    model = _model_for_settings(settings)

    mode = _normalise_mode(

        settings,

    platform_context = None
    platform_system_messages = []

    if mode == "multiplayer":

            require_multiplayer=True
        )

        platform_system_messages = [{
            "role": "system",
            "content": _multiplayer_integration_platform_message(
                platform_context
            )
        }]

    current_version = int(game["active_version"])

    from .storage import get_version

    current = get_version(
        game["id"],
        current_version
    )

    if not current or not current.get("html"):
        raise ValueError("Active game version is missing or empty.")

    instruction = {
        "original_prompt": game.get("prompt", ""),
        "settings": settings,
        "player_mode": mode,
        "change_request": change_request,
        "current_version": current_version,
        "current_game_html": current["html"],
        "required_behavior": {
            "return_complete_updated_html": True,
            "preserve_game_identity_and_all_working_features": True,
            "preserve_player_mode_and_multiplayer_protocol": True,
            "preserve_existing_adapter_and_safe_points": True,
            "preserve_all_gameplay_state_and_progress": True,
            "do_only_the_requested_change": True,
            "do_not_add_bonus_or_polish_features": True,
            "do_not_add_unrequested_progression_or_levels": True,
            "avoid_duplicate_if_else_blocks_and_handlers": True,
            "keep_installed_multiplayer_bridge_unchanged": True,
            "return_unchanged_html_if_change_is_unsafe": True
        }
    }

    messages = [
        {
            "role": "system",
            "content": SYSTEM_EVOLVE + "\n\n" + SYSTEM_NO_COMMENTS
        }
    ]

    if mode == "multiplayer":
        messages.extend(platform_system_messages)
        messages.append({
            "role": "system",
            "content": MULTIPLAYER_API_CONTRACT
        })

    messages.append({
        "role": "user",
        "content": json.dumps(instruction, ensure_ascii=False)
    })

    html = _call_html_model(
        messages,
        instruction,
        mode,
        "Game evolution",
        temperature=0.15,
        extra_system_messages=platform_system_messages,
        model=model
    )

    html = _prepare_html(html, mode)

    if mode == "multiplayer":
        html = _ensure_multiplayer_gameplay(
            html,
            game.get("prompt", ""),
            settings,
            {"player_mode": mode},
            stage="background-evolution",
            platform_context=platform_context,
            model=model
        )

    html, validation = _validate_with_ai_repair(
        html,
        "Upgrade failed validation",
        game.get("prompt", ""),
        settings,
        {"player_mode": mode},
        mode,
        platform_context=platform_context,
        stage="background-evolution",
        model=model
    )

    return {
        "html": html,
        "summary": "AI evolution: " + str(change_request)[:160],
        "validation": validation
    }

    if not current or not current.get("html"):
        raise ValueError("The active game version is missing or empty.")

    settings = _load_game_settings(game)
    model = _model_for_settings(settings)

    mode = _normalise_mode(
        game.get("game_mode"),
        settings,
        prefer_settings=False
    )
    platform_context = None
    platform_system_messages = []

    if mode == "multiplayer":
        platform_context = _load_platform_project_context(
            require_multiplayer=True
        )

        platform_system_messages = [{
            "role": "system",
            "content": _multiplayer_integration_platform_message(
                platform_context
            )
        }]

    instruction = {
        "original_request": game.get("prompt", ""),
        "settings": settings,
        "player_mode": mode,
        "current_game_html": current["html"],
        "task": (
            "Audit this exact game against the original request. "
            "Fix actual missing, broken, or incomplete requested features "
            "without removing anything that already works. "
            "Return the complete corrected HTML document."
        ),
        "required_checks": [
            "All requested controls and mechanics work.",
            "No mechanic exists unless requested or strictly necessary for a requested feature.",
            "No unrequested levels, stages, waves, rounds, level-up or level-down actions, XP, currency, upgrades, shops, power-ups, bosses, achievements, timers, or difficulty systems have been added.",
            "Menus and waiting screens transition correctly.",
            "No button or control is decorative or inert.",
            "There are no duplicated identical if/else blocks or repeated conditions performing the same operation.",
            "There are no duplicate event listeners, input handlers, timers, or animation loops.",
            "Gameplay state is preserved across compatible background upgrades."
        ]
    }

    messages = [
        {
            "role": "system",
            "content": SYSTEM_QA + "\n\n" + SYSTEM_NO_COMMENTS
        }
    ]

    if mode == "multiplayer":
        instruction["multiplayer_checks"] = [
            "Use only the public AIGameMultiplayer on and send methods.",
            "The installed platform bridge must remain present and unchanged.",
            "Use only the verified `connection`, `players`, `player_joined`, and `state` event channels. The first three are host/backend-to-game notifications; `state` is the shared gameplay data channel in both directions.",
            "Use mp.connected for status text; do not invent connected/disconnected events or any additional event names, create connection UI, or create another connection flow.",
            "Subscribe to the actual players roster event.",
            "Identify distinct players by stable player IDs.",
            "Start only when the plan's required roster is ready, and do so exactly once.",
            "Handle authoritative roster updates and late joins; do not assume leave or reconnect APIs exist.",
            "Synchronize actual gameplay state and render remote players without echoing received events.",
            "Never access the parent frame or recreate the platform bridge."
        ]
        messages.extend(platform_system_messages)
        messages.append({
            "role": "system",
            "content": MULTIPLAYER_API_CONTRACT
        })

    messages.append({
        "role": "user",
        "content": json.dumps(instruction, ensure_ascii=False)
    })

    html = _call_html_model(
        messages,
        instruction,
        mode,
        "Double-check",
        temperature=0.1,
        extra_system_messages=platform_system_messages,
        model=model
    )

    html = _prepare_html(html, mode)

    if mode == "multiplayer":
        html = _ensure_multiplayer_gameplay(
            html,
            game.get("prompt", ""),
            settings,
            {"player_mode": mode},
            stage="double-check",
            platform_context=platform_context,
            model=model
        )

    html, validation = _validate_with_ai_repair(
        html,
        "Double-check failed validation",
        game.get("prompt", ""),
        settings,
        {"player_mode": mode},
        mode,
        platform_context=platform_context,
        stage="double-check",
        model=model
    )

    return {
        "html": html,
        "summary": "AI Double-check: audited and repaired against the original request.",
        "validation": validation
    }
